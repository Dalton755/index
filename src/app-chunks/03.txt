function Orders({ data, customerNames, onNew, onOpen, showValue = true }: { data: AppData; customerNames: Map<string,string>; onNew?: () => void; onOpen: (id: string) => void; showValue?: boolean }) {
  const [query, setQuery] = useState('')
  const [filter, setFilter] = useState<'all' | 'orcamento' | 'running' | 'done'>('all')
  const visible = useMemo(() => data.orders.filter(o => {
    const customer = customerNames.get(o.cliente_id) ?? ''
    const matches = `OS ${o.numero} ${customer} ${o.titulo}`.toLowerCase().includes(query.toLowerCase())
    if (!matches) return false
    if (filter === 'orcamento') return ['rascunho','orcamento','aguardando_aprovacao'].includes(o.status)
    if (filter === 'running') return ['aprovada','agendada','em_deslocamento','em_execucao','pausada'].includes(o.status)
    if (filter === 'done') return o.status === 'concluida'
    return true
  }), [data.orders, customerNames, query, filter])
  return <section className="page-section"><div className="page-heading"><div><span>OPERAÇÃO</span><h1>Ordens de Serviço</h1></div>{onNew && <button className="square-add" onClick={onNew}><Icon name="plus"/></button>}</div><label className="search"><Icon name="search"/><input value={query} onChange={e=>setQuery(e.target.value)} placeholder="Buscar por cliente, OS ou serviço"/></label><div className="filter-row"><button className={filter==='all'?'selected':''} onClick={()=>setFilter('all')}>Todas</button><button className={filter==='orcamento'?'selected':''} onClick={()=>setFilter('orcamento')}>Orçamentos</button><button className={filter==='running'?'selected':''} onClick={()=>setFilter('running')}>Em andamento</button><button className={filter==='done'?'selected':''} onClick={()=>setFilter('done')}>Concluídas</button></div>{visible.length ? <div className="stack">{visible.map(o => <OrderCard key={o.id} order={o} customerName={customerNames.get(o.cliente_id) ?? 'Cliente'} showValue={showValue} onOpen={() => onOpen(o.id)}/>)}</div> : <EmptyState title="Nenhuma OS encontrada" text="Nenhuma ordem corresponde aos filtros atuais."/>}</section>
}
