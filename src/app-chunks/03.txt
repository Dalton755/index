function Finance({ data, customerNames }: { data: AppData; customerNames: Map<string,string> }) {
  const month = new Date().getMonth(); const year = new Date().getFullYear()
  const paidPayments = data.payments.filter(p=>p.status==='pago')
  const monthPaid = paidPayments.filter(p=>{const d=new Date(p.pago_em ?? p.created_at);return d.getMonth()===month&&d.getFullYear()===year}).reduce((s,p)=>s+Number(p.valor),0)
  const totalOrders = data.orders.filter(o=>o.status!=='cancelada').reduce((s,o)=>s+Number(o.total ?? 0),0)
  const totalPaid = paidPayments.reduce((s,p)=>s+Number(p.valor),0)
  const receivable = Math.max(totalOrders-totalPaid,0)

  return <section className="page-section"><div className="page-heading"><div><span>RESULTADO</span><h1>Financeiro</h1></div></div><div className="finance-hero"><span>Recebido neste mês</span><strong>{money.format(monthPaid)}</strong><small>somente pagamentos confirmados</small></div><div className="metrics-grid two"><div className="metric-card"><span>Recebido total</span><strong>{money.format(totalPaid)}</strong><small>{paidPayments.length} pagamentos</small></div><div className="metric-card"><span>A receber</span><strong>{money.format(receivable)}</strong><small>saldo previsto das OS</small></div></div><section className="section-block inner"><div className="section-title"><div><span>ÚLTIMAS MOVIMENTAÇÕES</span><h2>Recebimentos</h2></div></div>{paidPayments.length ? paidPayments.slice(0,5).map(p=>{ const order=orderMap.get(p.ordem_id); return <div className="payment-row" key={p.id}><div className="payment-icon"><Icon name="check"/></div><div><strong>{order ? customerNames.get(order.cliente_id) ??
