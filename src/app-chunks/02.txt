function Onboarding({ user, onDone }: { user: User; onDone: () => Promise<void> }) {
  const [companyName, setCompanyName] = useState('')
  const [phone, setPhone] = useState('')
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState('')

  async function submit(event: FormEvent) {
    event.preventDefault(); setBusy(true); setError('')
    try {
      const { data: company, error: companyError } = await supabase.from('empresas').insert({ nome: companyName, telefone: phone || null, email: user.email ?? null, segmento: 'os', created_by: user.id }).select('id').single()
      if (companyError) throw companyError
      const fullName = String(user.user_metadata?.full_name || user.user_metadata?.name || '').trim() || null
      const { error: memberError } = await supabase.from('membros').update({ nome: fullName, email: user.email ?? null }).eq('empresa_id', company.id).eq('user_id', user.id)
      if (memberError) throw memberError
      await onDone()
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Não foi possível criar a empresa.')
    } finally { setBusy(false) }
  }

  return <div className="auth-shell onboarding-shell"><section className="auth-card">
    <div className="auth-brand"><div className="brand-mark large">N</div><div><strong>Nethanel OS</strong><span>Configuração inicial</span></div></div>
    <div className="auth-copy"><span>PRIMEIRO ACESSO</span><h1>Vamos preparar sua operação.</h1><p>Você será cadastrado automaticamente como administrador dessa empresa.</p></div>
    <form onSubmit={submit}>
      <label>Nome da empresa ou profissional<input required autoFocus value={companyName} onChange={e => setCompanyName(e.target.value)} placeholder="Ex.: Clima Tech Serviços"/></label>
      <label>WhatsApp / telefone<input value={phone} onChange={e => setPhone(e.target.value)} inputMode="tel" placeholder="(11) 99999-9999"/></label>
      {error && <div className="form-alert error">{error}</div>}
      <button className="wide-primary" disabled={busy}>{busy ? 'Configurando...' : 'Começar a usar'}</button>
    </form>
  </section></div>
}

function EmptyState({ title, text }: { title: string; text: string }) {
  return <div className="empty-state"><div className="empty-icon"><Icon name="orders"/></div><strong>{title}</strong><span>{text}</span></div>
}

function OrderCard({ order, customerName, onOpen, showValue = true }: { order: Order; customerName: string; onOpen?: () => void; showValue?: boolean }) {
  const schedule = order.agendado_inicio ? new Date(order.agendado_inicio) : null
  const meta = statusMeta[order.status] ?? statusMeta.rascunho
  return <article className={`order-card ${onOpen ? 'clickable-card' : ''}`} onClick={onOpen} role={onOpen ? 'button' : undefined} tabIndex={onOpen ? 0 : undefined} onKeyDown={e => { if (onOpen && (e.key === 'Enter' || e.key === ' ')) onOpen() }}>
    <div className="order-head"><div><span className="order-code">OS #{String(order.numero).padStart(4, '0')}</span><h3>{customerName}</h3></div><span className={`status ${meta.className}`}>{meta.label}</span></div>
    <p>{order.titulo}</p>
    <div className="order-meta"><span><Icon name="calendar"/> {schedule ? dateFmt.format(schedule) : 'Sem data'}</span><span><Icon name="clock"/> {schedule ? timeFmt.format(schedule) : '--:--'}</span>{showValue && <strong>{money.format(Number(order.total ?? 0))}</strong>}</div>
  </article>
}
