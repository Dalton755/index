function Onboarding({ user, onDone }: { user: User; onDone: () => Promise<void> }) {
  const [companyName, setCompanyName] = useState('')
  const [phone, setPhone] = useState('')
  const [busy, setBusy] = useState(false)
  const [error, setError] = useState('')

  async function submit(event: FormEvent) {
    event.preventDefault(); setBusy(true); setError('')
    try {
      const { data: company, error: companyError } = await supabase.from('empresas').insert({ nome: companyName, telefone: phone || null, email: user.email ?? null, segmento: 'os', created_by: user.id }).select('id').single()
      if (companyError) throw companyError
      const fullName = String(user.user_metadata?.full_name || user.user_metadata?.name || '').trim() || null
      const { error: memberError } = await supabase.from('membros').update({ nome: fullName, email: user.email ?? null }).eq('empresa_id', company.id).eq('user_id', user.id)
      if (memberError) throw memberError
      await onDone()
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Não foi possível criar a empresa.')
    } finally { setBusy(false) }
  }

  return <div className="auth-shell onboarding-shell"><section className="auth-card">
    <div className="auth-brand"><div className="brand-mark large">N</div><div><strong>Nethanel OS</strong><span>Configuração inicial</span></div></div>
    <div className="auth-copy"><span>PRIMEIRO ACESSO</span><h1>Vamos preparar sua operação.</h1><p>Você será cadastrado automaticamente como administrador dessa empresa.</p></div>
    <form onSubmit={submit}>
      <label>Nome da empresa ou profissional<input required autoFocus value={companyName} onChange={e => setCompanyName(e.target.value)} placeholder="Ex.: Clima Tech Serviços"/></label>
      <label>WhatsApp / telefone<input value={phone} onChange={e => setPhone(e.target.value)} inputMode="tel" placeholder="(11) 99999-9999"/></label>
      {error && <div className="form-alert error">{error}</div>}
      <button className="wide-primary" disabled={busy}>{busy ? 'Configurando...' : 'Começar a usar'}</button>
    </form>
  </section></div>
}

  return <article className={`order-card ${onOpen ? 'clickable-card' : ''}`} onClick={onOpen} role={onOpen ? 'button' : undefined} tabIndex={onOpen ? 0 : undefined} onKeyDown={e => { if (onOpen && (e.key === 'Enter' || e.key === ' ')) onOpen() }}>

function Dashboard({ member, data, customerNames, onNew, onAgenda, onOpen }: { member: Member; data: AppData; customerNames: Map<string, string>; onNew: () => void; onAgenda: () => void; onOpen: (id: string) => void }) {
  const todayKey = localDateKey(new Date())
  const month = new Date().getMonth(); const year = new Date().getFullYear()
  const activeOrders = data.orders.filter(o => !['concluida', 'cancelada'].includes(o.status))
  const todayOrders = data.orders.filter(o => o.agendado_inicio && localDateKey(o.agendado_inicio) === todayKey)
  const paid = data.payments.filter(p => p.status === 'pago').reduce((sum, p) => sum + Number(p.valor), 0)
  const openTotal = data.orders.filter(o => o.status !== 'cancelada').reduce((sum, o) => sum + Number(o.total ?? 0), 0)
  const receivable = Math.max(openTotal - paid, 0)
  const monthPaid = data.payments.filter(p => { if (p.status !== 'pago') return false; const d = new Date(p.pago_em ?? p.created_at); return d.getMonth() === month && d.getFullYear() === year }).reduce((sum, p) => sum + Number(p.valor), 0)
  const dateLabel = fullDateFmt.format(new Date()).toUpperCase()
  const firstName = member.nome?.split(' ')[0] || 'profissional'

  return <>
    <section className="hero"><div><span className="eyebrow">{dateLabel}</span><h1>Olá, {firstName}.</h1><p>{activeOrders.length ? `${activeOrders.length} OS precisam da sua atenção.` : 'Sua operação está sob controle.'}</p></div><button className="primary-action" onClick={onNew}><Icon name="plus"/> Nova OS</button></section>
    <section className="metrics-grid"><div className="metric-card"><span>OS abertas</span><strong>{activeOrders.length}</strong><small>{todayOrders.length} para hoje</small></div><div className="metric-card"><span>A receber</span><strong>{money.format(receivable)}</strong><small>valor previsto</small></div><div className="metric-card"><span>Hoje</span><strong>{money.format(todayOrders.reduce((s,o)=>s+Number(o.total ?? 0),0))}</strong><small>{todayOrders.length} serviços</small></div><div className="metric-card accent"><span>Recebido no mês</span><strong>{money.format(monthPaid)}</strong><small>pagamentos confirmados</small></div></section>
    <section className="section-block"><div className="section-title"><div><span>PRÓXIMOS SERVIÇOS</span><h2>Agenda de hoje</h2></div><button onClick={onAgenda}>Ver agenda</button></div>{todayOrders.length ? <div className="timeline">{todayOrders.slice(0,3).map((o, index) => { const schedule = new Date(o.agendado_inicio as string); const meta = statusMeta[o.status]; return <div className="timeline-item" key={o.id} onClick={() => onOpen(o.id)}><time>{timeFmt.format(schedule)}</time><div className="timeline-line"><i/></div><div className="timeline-content"><strong>{customerNames.get(o.cliente_id) ?? 'Cliente'}</strong><span>{o.titulo}</span><b className={index % 2 ? 'blue' : ''}>{meta.label}</b></div></div> })}</div> : <EmptyState title="Agenda livre hoje" text="Crie uma OS e escolha data e horário para aparecer aqui."/>}</section>
    <section className="section-block"><div className="section-title"><div><span>ORDENS DE SERVIÇO</span><h2>Precisam da sua atenção</h2></div></div>{activeOrders.length ? <div className="stack">{activeOrders.slice(0,3).map(o => <OrderCard key={o.id} order={o} customerName={customerNames.get(o.cliente_id) ?? 'Cliente'} onOpen={() => onOpen(o.id)}/>)}</div> : <EmptyState title="Nenhuma OS aberta" text="Sua próxima ordem de serviço aparecerá aqui."/>}</section>
  </>
}
