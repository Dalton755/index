v><div><span>Pendente</span><strong>{money.format(data.financeiro.pendente)}</strong></div></div>}
    {canApprove?<div className="portal-approval"><span>APROVAÇÃO DO ORÇAMENTO</span><h2>Está de acordo com o serviço?</h2><label>Seu nome<input value={name} onChange={e=>setName(e.target.value)} placeholder="Nome completo"/></label>{error&&<div className="form-alert error">{error}</div>}<button disabled={busy} onClick={approve}><Icon name="check"/>{busy?'Confirmando...':'Aprovar orçamento'}</button><small>Ao aprovar, seu nome, data e horário ficam registrados na OS.</small></div>:<div className="portal-approved"><Icon name="check"/><div><strong>{data.ordem.aprovado_por_nome?'Orçamento aprovado':'OS em andamento'}</strong><span>{data.ordem.aprovado_em?`Aprovado por ${data.ordem.aprovado_por_nome} em ${dateTimeFmt.format(new Date(data.ordem.aprovado_em))}`:'Acompanhe o status acima.'}</span></div></div>}
    {data.ordem.assinatura_em&&<div className="portal-approved"><Icon name="check"/><div><strong>Serviço assinado</strong><span>{data.ordem.assinatura_nome} • {dateTimeFmt.format(new Date(data.ordem.assinatura_em))}</span></div></div>}
    <footer>Gerenciado por <strong>Nethanel OS</strong></footer>
  </section></div>
}

function OperationalApp({ session, member, company, reloadProfile }: { session: Session; member: Member; company: Company; reloadProfile: () => Promise<void> }) {
  const [tab,setTab]=useState<Tab>('home'); const [newOrder,setNewOrder]=useState(false); const [newCustomer,setNewCustomer]=useState(false); const [selectedOrderId,setSelectedOrderId]=useState<string|null>(null); const [loading,setLoading]=useState(true); const [error,setError]=useState(''); const [data,setData]=useState<AppData>({customers:[],orders:[],payments:[]})
  const canManage=member.papel==='admin'||member.papel==='gestor'; const canFinance=canManage||member.papel==='financeiro'; const showValue=member.papel!=='tecnico'
  const loadData=useCallback(async()=>{setError('');const [customersRes,ordersRes,paymentsRes]=await Promise.all([supabase.from('clientes').select('id,nome,telefone,whatsapp,email,created_at').eq('empresa_id',company.id).eq('ativo',true).order('nome'),supabase.from('ordens_servico').select('id,numero,cliente_id,titulo,descricao,status,agendado_inicio,tecnico_responsavel,total,created_at').eq('empresa_id',company.id).order('created_at',{ascending:false}),supabase.from('pagamentos').select('id,ordem_id,valor,metodo,status,pago_em,created_at').eq('empresa_id',company.id).order('created_at',{ascending:false})]);const firstError=customersRes.error||ordersRes.error||paymentsRes.error;if(firstError){setError(firstError.message)}else{setData({customers:(customersRes.data??[]) as Customer[],orders:(ordersRes.data??[]).map(o=>({...o,numero:Number(o.numero),total:o.total===null?null:Number(o.total)})) as Order[],payments:(paymentsRes.data??[]).map(p=>({...p,valor:Number(p.valor)})) as Payment[]})}setLoading(false)},[company.id])
  useEffect(()=>{void loadData()},[loadData])
  const customerNames=useMemo(()=>new Map(data.customers.map(c=>[c.id,c.nome])),[data.customers])
  const orderTotals=useMemo(()=>{const map=new Map<string,{count:number,total:number}>();data.orders.forEach(o=>{const v=map.get(o.cliente_id)??{count:0,total:0};v.count++;v.total+=Number(o.total??0);map.set(o.cliente_id,v)});return map},[data.orders])
  async function logout(){await supabase.auth.signOut();await reloadProfile()}
  const goTeam=()=>{setSelectedOrderId(null);setTab('team')}
  if(loading)return <LoadingScreen/>
  if(selectedOrderId){const selectedOrder=data.orders.find(o=>o.id===selectedOrderId);const customer=selectedOrder?data.customers.find(c=>c.id===selectedOrder.cliente_id):undefined;return <div className="app-shell"><Header company={company} member={member} userId={session.user.id} onOpenOrder={setSelectedOrderId} onTeam={canManage?goTeam:undefined} onLogout={logout}/><OrderDetailView orderId={selectedOrderId} companyId={company.id} userId={session.user.id} customer={customer} member={member} onBack={()=>setSelectedOrderId(null)} onChanged={loadData}/></div>}
  return <div className="app-shell"><Header company={company} member={member} userId={session.user.id} onOpenOrder={setSelectedOrderId} onTeam={canManage?goTeam:undefined} onLogout={logout}/>{error&&<div className="global-error">{error}<button onClick={()=>void loadData()}>Tentar novamente</button></div>}<main>
    {tab==='home'&&<Dashboard member={member} data={data} customerNames={customerNames} onNew={canManage?()=>setNewOrder(true):undefined} onAgenda={()=>setTab('agenda')} onOpen={setSelectedOrderId}/>}
    {tab==='orders'&&<Orders data={data} customerNames={customerNames} onNew={canManage?()=>setNewOrder(true):undefined} showValue={showValue} onOpen={setSelectedOrderId}/>}
    {tab==='customers'&&canManage&&<Customers data={data} orderTotals={orderTotals} onNew={()=>setNewCustomer(true)}/>}
    {tab==='agenda'&&<Agenda data={data} customerNames={customerNames} showValue={showValue} onOpen={setSelectedOrderId}/>}
    {tab==='finance'&&canFinance&&<Finance data={data} customerNames={customerNames}/>}
    {tab==='team'&&canManage&&<TeamPage companyId={company.id} companyName={company.nome} currentUserId={session.user.id}/>}
  </main>{canManage&&<button className="floating-add" onClick={()=>setNewOrder(true)} aria-label="Nova OS"><Icon name="plus"/></button>}<BottomNav tab={tab} onChange={setTab} member={member}/>{canManage&&newOrder&&<NewOrderModal companyId={company.id} userId={session.user.id} customers={data.customers} onClose={()=>setNewOrder(false)} onCreated={loadData}/>} {canManage&&newCustomer&&<NewCustomerModal companyId={company.id} userId={session.user.id} onClose={()=>setNewCustomer(false)} onCreated={loadData}/>}</div>
}

export default function App() {
  const searchParams = new URLSearchParams(window.location.search)
  const portalToken = searchParams.get('os')
  const inviteParam = searchParams.get('invite')
  if (inviteParam && localStorage.getItem('zelo_invite_token') !== inviteParam) localStorage.setItem('zelo_invite_token', inviteParam)
  const [session,setSession]=useState<Session|null>(null); const [member,setMember]=useState<Member|null>(null); const [company,setCompany]=useState<Company|null>(null); const [loading,setLoading]=useState(true); const [inviteError,setInviteError]=useState('')
  const loadProfile=useCallback(async(currentSession?:Session|null)=>{const activeSession=currentSession===undefined?(await supabase.auth.getSession()).data.session:currentSession;setSession(activeSession);if(!activeSession){setMember(null);setCompany(null);setInviteError('');setLoading(false);return}const pendingInvite=localStorage.getItem('zelo_invite_token');if(pendingInvite){const {error:acceptError}=await supabase.rpc('aceitar_convite_equipe',{p_token:pendingInvite});if(!acceptError){localStorage.removeItem('zelo_invite_token');setInviteError('');const url=new URL(window.location.href);url.searchParams.delete('invite');window.history.replaceState({},'',url.pathname+url.search)}else{setInviteError(acceptError.message)}}const {data:membership,error:memberError}=await supabase.from('membros').select('id,empresa_id,user_id,nome,email,papel').eq('user_id',activeSession.user.id).eq('ativo',true).limit(1).maybeSingle();if(memberError){console.error(memberError);setLoading(false);return}if(!membership){setMember(null);setCompany(null);setLoading(false);return}const {data:companyData,error:companyError}=await supabase.from('empresas').select('id,nome,segmento').eq('id',membership.empresa_id).single();if(companyError){console.error(companyError);setLoading(false);return}setMember(membership as Member);setCompany(companyData as Company);setLoading(false)},[])
  useEffect(()=>{if(portalToken){setLoading(false);return}void loadProfile();const {data:{subscription}}=supabase.auth.onAuthStateChange((_event,newSession)=>{setLoading(true);void loadProfile(newSession)});return()=>subscription.unsubscribe()},[loadProfile,portalToken])
  if(portalToken)return <PublicPortal token={portalToken}/>
  if(loading)return <LoadingScreen/>
  if(!session)return <AuthScreen/>
  if(inviteError&&!member)return <div className="auth-shell"><section className="auth-card"><div className="brand-mark large">N</div><div className="auth-copy"><span>CONVITE DE EQUIPE</span><h1>Não foi possível aceitar o convite.</h1><p>{inviteError}</p></div><button className="wide-primary" onClick={async()=>{localStorage.removeItem('zelo_invite_token');setInviteError('');await supabase.auth.signOut()}}>Entrar com outro e-mail</button></section></div>
  if(!member||!company)return <Onboarding user={session.user} onDone={async()=>{setLoading(true);await loadProfile(session)}}/>
  return <OperationalApp session={session} member={member} company={company} reloadProfile={async()=>{setLoading(true);await loadProfile()}}/>
}
