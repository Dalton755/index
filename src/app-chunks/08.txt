;const customer=selectedOrder?data.customers.find(c=>c.id===selectedOrder.cliente_id):undefined;return <div className="app-shell"><Header company={company} onLogout={logout}/><OrderDetailView orderId={selectedOrderId} companyId={company.id} userId={session.user.id} customer={customer} member={member} onBack={()=>setSelectedOrderId(null)} onChanged={loadData}/></div>}
  return <div className="app-shell"><Header company={company} onLogout={logout}/>{error&&<div className="global-error">{error}<button onClick={()=>void loadData()}>Tentar novamente</button></div>}<main>{tab==='home'&&<Dashboard member={member} data={data} customerNames={customerNames} onNew={()=>setNewOrder(true)} onAgenda={()=>setTab('agenda')} onOpen={setSelectedOrderId}/>} {tab==='orders'&&<Orders data={data} customerNames={customerNames} onNew={()=>setNewOrder(true)} onOpen={setSelectedOrderId}/>} {tab==='customers'&&<Customers data={data} orderTotals={orderTotals} onNew={()=>setNewCustomer(true)}/>} {tab==='agenda'&&<Agenda data={data} customerNames={customerNames} onOpen={setSelectedOrderId}/>} {tab==='finance'&&<Finance data={data} customerNames={customerNames}/>}</main><button className="floating-add" onClick={()=>setNewOrder(true)} aria-label="Nova OS"><Icon name="plus"/></button><BottomNav tab={tab} onChange={setTab}/>{newOrder&&<NewOrderModal companyId={company.id} userId={session.user.id} customers={data.customers} onClose={()=>setNewOrder(false)} onCreated={loadData}/>} {newCustomer&&<NewCustomerModal companyId={company.id} userId={session.user.id} onClose={()=>setNewCustomer(false)} onCreated={loadData}/>}</div>
}

export default function App() {
  const portalToken = new URLSearchParams(window.location.search).get('os')
  const [session,setSession]=useState<Session|null>(null); const [member,setMember]=useState<Member|null>(null); const [company,setCompany]=useState<Company|null>(null); const [loading,setLoading]=useState(true)
  const loadProfile=useCallback(async(currentSession?:Session|null)=>{const activeSession=currentSession===undefined?(await supabase.auth.getSession()).data.session:currentSession;setSession(activeSession);if(!activeSession){setMember(null);setCompany(null);setLoading(false);return}const {data:membership,error:memberError}=await supabase.from('membros').select('id,empresa_id,nome,email,papel').eq('user_id',activeSession.user.id).eq('ativo',true).limit(1).maybeSingle();if(memberError){console.error(memberError);setLoading(false);return}if(!membership){setMember(null);setCompany(null);setLoading(false);return}const {data:companyData,error:companyError}=await supabase.from('empresas').select('id,nome,segmento').eq('id',membership.empresa_id).single();if(companyError){console.error(companyError);setLoading(false);return}setMember(membership as Member);setCompany(companyData as Company);setLoading(false)},[])
  useEffect(()=>{if(portalToken){setLoading(false);return}void loadProfile();const {data:{subscription}}=supabase.auth.onAuthStateChange((_event,newSession)=>{setLoading(true);void loadProfile(newSession)});return()=>subscription.unsubscribe()},[loadProfile,portalToken])
  if(portalToken)return <PublicPortal token={portalToken}/>
  if(loading)return <LoadingScreen/>
  if(!session)return <AuthScreen/>
  if(!member||!company)return <Onboarding user={session.user} onDone={async()=>{setLoading(true);await loadProfile(session)}}/>
  return <OperationalApp session={session} member={member} company={company} reloadProfile={async()=>{setLoading(true);await loadProfile()}}/>
}
